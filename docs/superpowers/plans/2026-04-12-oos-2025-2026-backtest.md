# OOS 2025 + 2026 YTD Backtest Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Run OOS backtests on all 15 trained embeddings models (PPO×5, A2C×5, SAC×5) for two new periods: 2025-01-01→2025-04-10 (full NLP) and 2026-01-01→2026-04-12 (price only, NLP zeros), then generate a combined report.

**Architecture:** Build two new parquet feature files using the EXISTING `data/processed/btc_4h_compressor.pkl` (no refit) and the existing `build_embedding_features` logic. For 2025: fetch real news from `edaschau/bitcoin_news` dataset + BTC 4h prices from Binance. For 2026: fetch prices only, fill all NLP columns with zeros. Both files share the exact same 95-column schema as `btc_4h_embedding_features.parquet`. Then run `run_oos_backtest` twice and produce a three-period comparison report (2024 vs 2025 vs 2026).

**Tech Stack:** Python 3.10+, ccxt (Binance), HuggingFace datasets, sentence-transformers (all-MiniLM-L6-v2), FinBERT, pandas/numpy, matplotlib, scipy (Mann-Whitney).

```python
"""Build BTC 4h embedding features for 2025-01-01 to 2025-04-10.

Uses existing PCA compressor — NO refit.
Output: data/processed/btc_4h_2025_embedding_features.parquet
"""
import logging
import sys
from pathlib import Path

import numpy as np
import pandas as pd

sys.path.insert(0, str(Path(__file__).parent.parent))

from src.data.price_collector import fetch_ohlcv
from src.features.technical import add_technical_indicators
from src.features.normalizer import rolling_zscore_normalize
from src.features.embedding_compressor import EmbeddingCompressor
from src.features.build_embedding_features import build_embedding_features
from src.data.news_collector import load_news
from src.data.news_preprocessor import preprocess_news

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
logger = logging.getLogger(__name__)

START = "2025-01-01"
END = "2025-04-10"
ASSET = "BTC/USDT"
TIMEFRAME = "4h"
COMPRESSOR_PATH = "data/processed/btc_4h_compressor.pkl"
OUTPUT_PATH = "data/processed/btc_4h_2025_embedding_features.parquet"


def main():
    # 1. Fetch prices
    logger.info(f"Fetching {ASSET} {TIMEFRAME} prices {START}→{END}")
    prices_df = fetch_ohlcv(ASSET, start=START, end=END, timeframe=TIMEFRAME)
    logger.info(f"Prices shape: {prices_df.shape}")

    # 2. Add technical indicators
    logger.info("Adding technical indicators")
    prices_df = add_technical_indicators(prices_df)

    # 3. Save raw_close before normalisation
    prices_df["raw_close"] = prices_df["close"]

    # 4. Normalize (rolling z-score, window=30)
    logger.info("Normalizing features")
    cols_to_norm = [c for c in prices_df.columns if c not in ("raw_close",)]
    prices_df[cols_to_norm] = rolling_zscore_normalize(prices_df[cols_to_norm])

    # 5. Load news 2025
    logger.info("Loading news 2025")
    news_df = load_news(
        "edaschau/bitcoin_news",
        start=START,
        end=END,
        split="train",
        title_col="title",
        text_col="article_text",
        date_col="date_time",
    )
    logger.info(f"Loaded {len(news_df)} news articles")

    # 6. Preprocess news → group by day
    logger.info("Preprocessing news")
    news_by_day = preprocess_news(news_df)
    logger.info(f"News days: {len(news_by_day)}")

    # 7. Load compressor (NO refit)
    logger.info(f"Loading compressor from {COMPRESSOR_PATH}")
    compressor = EmbeddingCompressor.load(COMPRESSOR_PATH)

    # 8. Build embedding features
    logger.info("Building embedding features")
    result = build_embedding_features(prices_df, news_by_day, compressor)
    logger.info(f"Result shape: {result.shape}")

    # 9. Verify schema matches training data (95 feature cols + 6 OHLCV)
    expected_feat_cols = 95
    feat_cols = [c for c in result.columns
                 if c.lower() not in ("open", "high", "low", "close", "volume", "raw_close")]
    assert len(feat_cols) == expected_feat_cols, \
        f"Expected {expected_feat_cols} feature cols, got {len(feat_cols)}: {feat_cols}"

    # 10. Save
    Path(OUTPUT_PATH).parent.mkdir(parents=True, exist_ok=True)
    result.to_parquet(OUTPUT_PATH)
    logger.info(f"Saved to {OUTPUT_PATH}")
    logger.info(f"Date range: {result.index.min()} → {result.index.max()}")


if __name__ == "__main__":
    main()
```

```python
"""Multi-period OOS comparison report: 2024 vs 2025 vs 2026 YTD."""
from __future__ import annotations

import logging
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from scipy.stats import mannwhitneyu

logger = logging.getLogger(__name__)

PERIODS = {
    "2024": "results/oos_2024_embeddings.csv",
    "2025 (NLP)": "results/oos_2025_embeddings.csv",
    "2026 YTD (no NLP)": "results/oos_2026_embeddings.csv",
}

BH_STATS = {
    "2024": {"sharpe": 0.7212, "total_return": 1.1854},
    "2025 (NLP)": None,   # computed at runtime
    "2026 YTD (no NLP)": None,
}


def _compute_bh(feature_path: str, test_start: str, test_end: str) -> dict:
    """Compute Buy & Hold metrics for a period."""
    import pandas as pd
    from src.eval.baselines import buy_and_hold
    df = pd.read_parquet(feature_path)
    start_ts = pd.Timestamp(test_start)
    end_ts = pd.Timestamp(test_end)
    if df.index.tz is not None:
        start_ts = start_ts.tz_localize("UTC") if start_ts.tzinfo is None else start_ts
        end_ts = end_ts.tz_localize("UTC") if end_ts.tzinfo is None else end_ts
    df = df.loc[start_ts:end_ts]
    if "raw_close" in df.columns:
        prices = df["raw_close"].values
    else:
        prices = df["close"].values
    bh = buy_and_hold(prices)
    return {
        "sharpe": bh["metrics"]["sharpe_ratio"],
        "total_return": bh["metrics"]["total_return"],
    }


def generate_multi_period_report(
    output_md: Path = Path("results/oos_multi_period_report.md"),
    figures_dir: Path = Path("results/figures/multi_period"),
) -> Path:
    output_md = Path(output_md)
    figures_dir = Path(figures_dir)
    figures_dir.mkdir(parents=True, exist_ok=True)
    output_md.parent.mkdir(parents=True, exist_ok=True)

    # Load all periods
    dfs = {}
    for period, csv_path in PERIODS.items():
        if Path(csv_path).exists():
            dfs[period] = pd.read_csv(csv_path)
        else:
            logger.warning(f"Missing: {csv_path}")

    # Compute Buy & Hold for 2025 and 2026
    bh_stats = dict(BH_STATS)
    period_features = {
        "2025 (NLP)": ("data/processed/btc_4h_2025_embedding_features.parquet", "2025-01-01", "2025-04-10"),
        "2026 YTD (no NLP)": ("data/processed/btc_4h_2026_embedding_features.parquet", "2026-01-01", "2026-04-12"),
    }
    for period, (fp, ts, te) in period_features.items():
        if Path(fp).exists():
            try:
                bh_stats[period] = _compute_bh(fp, ts, te)
            except Exception as e:
                logger.warning(f"Could not compute B&H for {period}: {e}")

    lines = ["# Embeddings 3-Agents — Multi-Period OOS Report\n"]
    lines.append("Periods: **2024** (in-distribution OOS) · **2025** (full NLP) · **2026 YTD** (price only, NLP=0)\n")
    lines.append(f"Models: 15 (PPO×5, A2C×5, SAC×5), trained on BTC/USDT 4h 2020–2023\n")

    # Per-period summary tables
    for period, df in dfs.items():
        summary = df.groupby("algorithm").agg(
            sharpe_mean=("sharpe", "mean"),
            sharpe_std=("sharpe", "std"),
            total_return_mean=("total_return", "mean"),
            total_return_std=("total_return", "std"),
            max_dd_mean=("max_drawdown", "mean"),
            calmar_mean=("calmar", "mean"),
        ).round(4)
        bh = bh_stats.get(period)
        lines.append(f"\n## {period}\n")
        if bh:
            lines.append(f"**Buy & Hold**: Sharpe={bh['sharpe']:.4f}, Total Return={bh['total_return']:.4f}\n")
        lines.append(summary.to_markdown())
        lines.append("\n")

    # Cross-period Sharpe comparison figure
    fig, axes = plt.subplots(1, len(dfs), figsize=(5 * len(dfs), 4), sharey=True)
    if len(dfs) == 1:
        axes = [axes]
    algos = ["PPO", "A2C", "SAC"]
    colors = {"PPO": "#4C72B0", "A2C": "#55A868", "SAC": "#C44E52"}
    for ax, (period, df) in zip(axes, dfs.items()):
        summary = df.groupby("algorithm")[["sharpe"]].agg(["mean", "std"])
        summary.columns = ["mean", "std"]
        means = [summary.loc[a, "mean"] if a in summary.index else 0 for a in algos]
        stds = [summary.loc[a, "std"] if a in summary.index else 0 for a in algos]
        ax.bar(algos, means, yerr=stds, capsize=6, color=[colors[a] for a in algos])
        bh = bh_stats.get(period)
        if bh:
            ax.axhline(bh["sharpe"], color="gray", linestyle="--", label=f"B&H {bh['sharpe']:.2f}")
            ax.legend(fontsize=8)
        ax.set_title(period, fontsize=10)
        ax.set_ylabel("Sharpe")
        ax.axhline(0, color="black", linewidth=0.5)
    fig.suptitle("OOS Sharpe by Period (mean ± std over 5 seeds)", fontsize=12)
    fig.tight_layout()
    fig.savefig(figures_dir / "sharpe_by_period.png", dpi=150)
    plt.close(fig)
    lines.append("\n## Figures\n")
    lines.append("- `figures/multi_period/sharpe_by_period.png` — Sharpe bar chart per period\n")

    # Mann-Whitney U for best algo across periods

    for a, b in combinations(period_list, 2):
```
